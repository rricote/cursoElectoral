categoria = s => s === 1 ? "Uninominal (1)" : s <= 5 ? "Pequeña (2-5)" : s <= 9 ? "Mediana (6-9)" : "Grande (10+)"
colorCat = ({"Uninominal (1)": "#c9c4bd", "Pequeña (2-5)": "#f2c6b4", "Mediana (6-9)": "#e07a5f", "Grande (10+)": "#9b2226"})
pct = v => v.toLocaleString("es-ES", {minimumFractionDigits: 1, maximumFractionDigits: 1}) + " %"
miles = d3.formatLocale({decimal: ",", thousands: ".", grouping: [3], currency: ["", " €"]}).format(",d")

// Dibuja un mapa de provincias. grupo = null → mapa general de escaños
function mapaSVG(eleccion, grupo) {
  const e = efectos[eleccion];
  const W = mapa.w, H = mapa.h + 10;
  const svg = d3.create("svg").attr("viewBox", `0 0 ${W} ${H}`).style("height", "282px").style("width", "auto");
  const relleno = id => {
    const s = e.escanos_prov[id];
    if (!grupo) return colorCat[categoria(s)];
    const d = grupo.prov[id];
    if (!d) return "#efedea";
    const color = grupo.k.startsWith("OTROS") ? d.c : grupo.c;
    return d3.interpolateRgb("#f3efe9", color)(0.3 + 0.7 * d.es / s);
  };
  const texto = id => grupo ? (grupo.prov[id]?.es ?? "") : e.escanos_prov[id];
  const titulo = id => {
    const s = e.escanos_prov[id];
    if (!grupo) return `${s} escaños`;
    const d = grupo.prov[id];
    return d ? `${d.es} de ${s} escaños${grupo.k.startsWith("OTROS") ? " · " + d.detalle : ""}` : `0 de ${s} escaños`;
  };
  const g = svg.append("g");
  for (const [id, path] of Object.entries(mapa.paths)) {
    g.append("path").attr("d", path).attr("fill", relleno(id)).attr("stroke", "#fff").attr("stroke-width", 1.5)
      .append("title").text(titulo(id));
  }
  for (const [id, [x, y, s]] of Object.entries(mapa.sq)) {
    g.append("rect").attr("x", x).attr("y", y).attr("width", s).attr("height", s).attr("rx", 6)
      .attr("fill", relleno(id)).attr("stroke", "#aaa").append("title").text(titulo(id));
  }
  const [x0, y0, x1, y1] = mapa.box;
  g.append("rect").attr("x", x0).attr("y", y0).attr("width", x1 - x0).attr("height", y1 - y0).attr("fill", "none").attr("stroke", "#ccc").attr("stroke-dasharray", "4 4");
  g.append("path").attr("d", mapa.paths["43"]).attr("fill", "none").attr("stroke", "#111").attr("stroke-width", 3);
  for (const [id, [x, y]] of Object.entries(mapa.lab)) {
    const t = texto(id);
    if (t === "") continue;
    const oscuro = !grupo ? e.escanos_prov[id] >= 10 : (grupo.prov[id] && grupo.prov[id].es / e.escanos_prov[id] > 0.45);
    g.append("text").attr("x", x).attr("y", y + 14).attr("text-anchor", "middle").attr("font-size", 46).attr("font-weight", 700)
      .attr("font-family", "Inter, sans-serif").attr("fill", oscuro ? "#fff" : "#222").attr("pointer-events", "none").text(t);
  }
  for (const [id, [x, y, s]] of Object.entries(mapa.sq)) {
    const t = texto(id);
    if (t === "") continue;
    g.append("text").attr("x", x + s / 2).attr("y", y + s / 2 + 14).attr("text-anchor", "middle").attr("font-size", 36).attr("font-weight", 700)
      .attr("font-family", "Inter, sans-serif").attr("fill", "#222").text(t);
  }
  return svg.node();
}

function celda(eleccion, k) {
  const e = efectos[eleccion];
  if (!k) {
    const orden = ["Uninominal (1)", "Pequeña (2-5)", "Mediana (6-9)", "Grande (10+)"];
    return html`<div class="celda">${mapaSVG(eleccion, null)}
      <div class="info"><div class="nombre">Escaños por provincia</div>
        <div class="num" style="color:#9b2226">350</div>
        ${orden.map(c => html`<div class="ley"><span style="background:${colorCat[c]}"></span>${c}</div>`)}</div></div>`;
  }
  const g = e.grupos.find(d => d.k === k);
  const dif = g.pct_es - g.pct;
  return html`<div class="celda">${mapaSVG(eleccion, g)}
    <div class="info"><div class="nombre" style="color:${g.k.startsWith("OTROS") ? "#333" : g.c}">${g.n}</div>
      <div class="num" style="color:${g.k.startsWith("OTROS") ? "#333" : g.c}">${g.es}</div><div>escaños</div>
      <div class="dato">${pct(g.pct)} de los votos</div>
      <div class="dato">${pct(g.pct_es)} de los escaños</div>
      <div class="dato" style="color:${dif >= 0 ? "#1b7f3b" : "#b3261e"};font-weight:700">${dif >= 0 ? "+" : ""}${pct(dif)}</div>
      <div class="dato gris">Escaño en ${Object.keys(g.prov).length} de 52 provincias</div></div></div>`;
}

cuadricula = (eleccion, ks) => html`<div class="cuadricula">${celda(eleccion, null)}${ks.map(k => celda(eleccion, k))}</div>`

La idea clave

El sistema español es proporcional, pero con 52 circunscripciones, la mayoría pequeñas. Eso produce tres efectos:

Premio a los grandes En las provincias de 2 a 5 escaños, el primer y el segundo partido se llevan casi todo.

Castigo a los terceros dispersos Un partido con el 10-15 % repartido por toda España se queda sin escaño en muchas provincias.

Ninguna penalización a los concentrados Los nacionalistas solo compiten en su territorio, donde son grandes, y sacan escaños a buen precio.

Es el efecto mecánico de Duverger. El efecto psicológico (el voto útil) refuerza a los grandes en las siguientes elecciones.

Abril de 2019

Abril de 2019 · PSOE, PP y Cs

cuadricula("2019-04", ["PSOE", "PP", "CS"])

El PSOE gana con el 28,7 % de los votos y saca el 35 % de los escaños: está presente en 51 de las 52 provincias. PP y Cs casi empatan en votos (16,7 % y 15,9 %) y el PP saca 9 escaños más: el segundo puesto en las provincias pequeñas vale mucho.

Abril de 2019 · UP, Vox y otros

cuadricula("2019-04", ["UP", "VOX", "OTROS"])

Vox, con el 10,3 % de los votos, solo saca escaño en 18 provincias: el voto disperso se pierde en las pequeñas. Los nacionalistas, en cambio, concentran sus escaños en Cataluña, el País Vasco, Navarra y Canarias. En el mapa de “otros” cada provincia tiene el color del partido que más escaños saca allí.

Vox pasa del 10,3 % al 15,1 % y de 24 a 52 escaños: al superar el umbral en muchas más provincias, sus escaños crecen mucho más que sus votos. El PP también recupera terreno.

El hundimiento de Ciudadanos es el mejor ejemplo del efecto mecánico: con el 6,8 % de los votos solo saca 10 escaños, casi todos en provincias grandes. Con un sistema proporcional puro habría tenido unos 24.

El índice de desproporcionalidad

Gallagher (1991) mide la distancia entre votos y escaños:

LSq = √( ½ · Σ (vi − ei)² )
  • v es el % de votos de cada partido y e, su % de escaños.
  • 0 = proporcionalidad perfecta. Cuanto más alto, más distorsión.
  • Como referencia: los Países Bajos suelen estar en torno a 1; el Reino Unido, con distritos uninominales, suele pasar de 10.

España está en un punto intermedio: es proporcional, pero con una distorsión moderada que favorece a los grandes y a los partidos concentrados.

Ganadores y perdedores del sistema

Ganan

  • El primer partido en cada elección (PSOE en 2019, PP en 2023).
  • El segundo, en menor medida.
  • Los partidos concentrados: PNV, EH Bildu, Junts, ERC, Navarra Suma, Teruel Existe.

Pierden

  • Los terceros y cuartos partidos estatales: UP, Vox (abril de 2019), Sumar y Vox (2023).
  • Sobre todo Ciudadanos en noviembre de 2019.
  • Los que no llegan a escaño: PACMA.

Consecuencias

  • Los nuevos partidos lo tienen difícil para consolidarse.
  • El voto útil empuja hacia los grandes.
  • Los nacionalistas tienen más peso del que indican sus votos y son decisivos para formar gobierno.

Resumen de la sesión

  1. El sistema es proporcional en teoría, pero las provincias pequeñas le dan efectos mayoritarios.
  2. Los mapas muestran que los grandes sacan escaño en casi todas las provincias; los terceros dispersos, solo en las grandes.
  3. El coste de un escaño varía muchísimo: de unos 20.000 votos (Teruel Existe, 2019) a más de 200.000 (Más País, 2019).
  4. El índice de Gallagher se sitúa entre 5 y 7: una desproporcionalidad moderada.
  5. Así se explica que España mantenga dos grandes partidos y que el multipartidismo nunca llegue a ser pleno.
