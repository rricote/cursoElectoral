function dibujaMapa(m, escanos, porId) {
  const svg = d3.create("svg").attr("viewBox", `0 0 ${m.w} ${m.h + 10}`).style("height", "400px").style("width", "auto");
  const g = svg.append("g");
  for (const [id, path] of Object.entries(m.paths)) {
    g.append("path").attr("d", path).attr("fill", colorCat[categoria(escanos[id])]).attr("stroke", "#fff").attr("stroke-width", porId ? 1.5 : 3)
      .append("title").text(`${porId ? porId[id] + ": " : ""}${escanos[id]} escaños`);
  }
  for (const [id, [x, y, s]] of Object.entries(m.sq)) {
    g.append("rect").attr("x", x).attr("y", y).attr("width", s).attr("height", s).attr("rx", 6).attr("fill", colorCat[categoria(escanos[id])]).attr("stroke", "#aaa");
    g.append("text").attr("x", x + s / 2).attr("y", y + s / 2 + 13).attr("text-anchor", "middle").attr("font-size", 34).attr("font-weight", 700).attr("font-family", "Inter, sans-serif").text(escanos[id]);
  }
  const [x0, y0, x1, y1] = m.box;
  g.append("rect").attr("x", x0).attr("y", y0).attr("width", x1 - x0).attr("height", y1 - y0).attr("fill", "none").attr("stroke", "#ccc").attr("stroke-dasharray", "4 4");
  for (const [id, [x, y]] of Object.entries(m.lab)) {
    if (escanos[id] == null) continue;
    g.append("text").attr("x", x).attr("y", y + 13).attr("text-anchor", "middle").attr("font-size", porId ? 30 : 40).attr("font-weight", 700)
      .attr("font-family", "Inter, sans-serif").attr("fill", escanos[id] >= 10 ? "#fff" : "#222").attr("pointer-events", "none").text(escanos[id]);
  }
  return svg.node();
}

function resumenCat(escanos) {
  const vals = Object.values(escanos);
  return html`<div style="font-family:Inter,sans-serif;font-size:14px;display:flex;flex-wrap:wrap;gap:4px 14px;align-items:center">
    ${ordenCat.map(c => { const n = vals.filter(s => categoria(s) === c).length;
      return html`<span style="display:inline-flex;align-items:center;gap:5px;opacity:${n ? 1 : .35}">
        <span style="display:inline-block;width:12px;height:12px;border-radius:3px;background:${colorCat[c]}"></span>${c}: <b>${n}</b></span>`; })}
    <span style="color:#555"><b>${vals.length}</b> circunscripciones</span></div>`;
}

La propuesta

1. Circunscripción La comunidad autónoma en lugar de la provincia: 17 circunscripciones más Ceuta y Melilla.

2. Ceuta y Melilla Siguen siendo uninominales: un escaño cada una.

3. Mínimo de 6 Cada comunidad parte de 6 escaños (102 en total). Los 246 restantes se reparten por población, como ahora.

4. Lo demás no cambia 350 diputados, fórmula D’Hondt y barrera del 3 %, aplicada ahora en cada comunidad.

He recalculado las cinco últimas elecciones generales con los votos reales y la población oficial de cada convocatoria, solo cambiando la circunscripción.

La idea de la propuesta es atacar la causa principal de la desproporcionalidad que vimos en la sesión 4: las provincias pequeñas. Al pasar a la comunidad autónoma, casi todas las circunscripciones tienen 8 escaños o más, y la barrera efectiva baja mucho. Se mantiene todo lo demás (350 diputados, D’Hondt y el 3 %) para ver el efecto aislado del cambio de circunscripción. El reparto entre comunidades se hace igual que la LOREG reparte hoy entre provincias: un mínimo fijo y el resto por población, con cociente y restos mayores.

function hemiciclo(partidos, campo, filas = 10, r0 = 0.4) {
  const total = d3.sum(partidos, p => p[campo]);
  const radios = d3.range(filas).map(i => r0 + (1 - r0) * i / (filas - 1));
  const sumR = d3.sum(radios);
  const porFila = radios.map(r => Math.round(total * r / sumR));
  porFila[filas - 1] += total - d3.sum(porFila);
  const asientos = [];
  radios.forEach((r, i) => {
    const n = porFila[i];
    for (let j = 0; j < n; j++) {
      const a = Math.PI - Math.PI * j / (n - 1);
      asientos.push({a, r, x: r * Math.cos(a), y: r * Math.sin(a)});
    }
  });
  asientos.sort((p, q) => q.a - p.a || p.r - q.r);
  let k = 0;
  for (const p of partidos.slice().sort((a, b) => posicion(a.k) - posicion(b.k)))
    for (let i = 0; i < p[campo]; i++) Object.assign(asientos[k++], {partido: p.n, c: p.c});
  return asientos;
}

Quién gana y quién pierde con la propuesta

Ganan

  • Los terceros y cuartos partidos estatales: Podemos/UP/Sumar, Ciudadanos, Vox.
  • IU en 2015: de 2 a 8 escaños.
  • Las comunidades uniprovinciales pequeñas y Madrid.

Pierden

  • PP y PSOE, entre 2 y 14 escaños por elección.
  • Los partidos de una sola provincia, como Teruel Existe.
  • Las comunidades con muchas provincias poco pobladas: Castilla y León, Castilla-La Mancha, Galicia.

Casi no cambian

  • Los nacionalistas (ERC, Junts, PNV, EH Bildu): ya compiten solo donde son fuertes.
  • La lógica de bloques: ningún bloque gana la mayoría absoluta gracias al cambio.

Para el debate

Argumentos a favor

  • El voto vale más parecido en toda España.
  • Menos voto útil forzado y más pluralismo.
  • Las comunidades ya son el gran nivel de gobierno territorial.

Argumentos en contra

  • Se pierde el vínculo entre el diputado y su provincia.
  • Las provincias rurales y despobladas pierden peso.
  • Más fragmentación: puede costar más formar gobierno.

Qué haría falta La provincia como circunscripción está en el artículo 68.2 de la Constitución. La propuesta exige una reforma constitucional (3/5 del Congreso y del Senado) y después cambiar la LOREG (mayoría absoluta del Congreso).

Es una propuesta personal y conviene presentarla con sus pros y sus contras. Una pregunta para abrir el debate: ¿qué preferís, que cada voto valga lo mismo o que cada provincia tenga sus propios diputados? Detalle técnico: el artículo 68 está en el Título III, así que su reforma sigue el procedimiento ordinario del artículo 167, no el agravado del 168.

Resumen de la sesión

  1. La propuesta cambia solo la circunscripción: de 52 provincias a 17 comunidades más Ceuta y Melilla, con un mínimo de 6 escaños.
  2. Desaparecen las circunscripciones pequeñas y la desproporcionalidad cae a la mitad o menos.
  3. Ganan los partidos estatales medianos y pierden algo PP y PSOE; los nacionalistas casi no cambian.
  4. Las comunidades con muchas provincias pequeñas pierden escaños; el mínimo de 6 sigue favoreciendo a las comunidades pequeñas.
  5. Requiere reformar la Constitución.
